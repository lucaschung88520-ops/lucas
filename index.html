# ============================================================ # 人工智慧與經濟研究 # 題目：人工智慧與技術進步對台灣勞動生產力及薪資成長之影響 # # 資料來源：AREMOS # 分析方法：資料清理、年增率、視覺化、OLS迴歸 # ============================================================ # ========================= # 1. 載入套件 # ========================= import pandas as pd import numpy as np import matplotlib.pyplot as plt import statsmodels.api as sm # ========================= # 2. 讀取 AREMOS CSV # ========================= # 請把 AREMOS 匯出的 CSV 檔案命名為： # aremos_data.csv df = pd.read_csv( "aremos_data.csv", encoding="utf-8-sig" ) print("原始資料：") print(df.head()) print("\n資料欄位：") print(df.columns.tolist()) # ========================= # 3. 資料清理 # ========================= # 第一欄設定為日期 date_column = df.columns[0] df[date_column] = pd.to_datetime( df[date_column], errors="coerce" ) # 將日期欄位命名為 Date df = df.rename( columns={date_column: "Date"} ) # 按照日期排序 df = df.sort_values("Date") # 設定日期為索引 df = df.set_index("Date") # ========================= # 4. 將資料轉成數字 # ========================= for column in df.columns: df[column] = pd.to_numeric( df[column], errors="coerce" ) # ========================= # 5. 檢查缺失值 # ========================= print("\n缺失值檢查：") print(df.isnull().sum()) # ========================= # 6. 移除缺失值 # ========================= df = df.dropna() print("\n清理後資料：") print(df.head()) # ============================================================ # 7. 設定研究變數 # ============================================================ # 請依照你的 AREMOS CSV 實際欄位名稱修改 # # 例如： # 勞動生產力 # 受僱員工薪資 # 工業生產指數 # # 如果你的欄位名稱不同，只需要修改下面三行。 PRODUCTIVITY = "勞動生產力" WAGE = "受僱員工薪資" INDUSTRIAL = "工業生產指數" # ========================= # 8. 計算年增率 # ========================= # 如果你的 AREMOS 資料是「月資料」 # 12 代表與去年同月比較 df["勞動生產力年增率"] = ( df[PRODUCTIVITY].pct_change(12) * 100 ) df["薪資年增率"] = ( df[WAGE].pct_change(12) * 100 ) df["工業生產年增率"] = ( df[INDUSTRIAL].pct_change(12) * 100 ) # 移除年增率計算產生的空值 df = df.dropna() # ========================= # 9. 顯示分析資料 # ========================= print("\n年增率資料：") print( df[ [ "勞動生產力年增率", "薪資年增率", "工業生產年增率" ] ].head() ) # ============================================================ # 10. 資料視覺化 # ============================================================ plt.figure(figsize=(12, 6)) plt.plot( df.index, df["勞動生產力年增率"], label="Labor Productivity Growth" ) plt.plot( df.index, df["薪資年增率"], label="Wage Growth" ) plt.axhline( y=0, linestyle="--" ) plt.title( "Taiwan Labor Productivity and Wage Growth" ) plt.xlabel("Date") plt.ylabel( "Year-over-Year Growth (%)" ) plt.legend() plt.grid(True) plt.tight_layout() plt.show() # ============================================================ # 11. OLS 基本迴歸分析 # ============================================================ # 研究模型： # # 薪資年增率 # = # α # + β1 × 勞動生產力年增率 # + β2 × 工業生產年增率 # + ε # # 研究假說： # # β1 > 0 # # 如果 β1 為正且具有統計顯著性， # 可以支持「勞動生產力提高與薪資成長具有正向關係」。 # X：解釋變數 X = df[ [ "勞動生產力年增率", "工業生產年增率" ] ] # Y：被解釋變數 Y = df["薪資年增率"] # 加入截距項 X = sm.add_constant(X) # 建立 OLS 模型 model = sm.OLS( Y, X ).fit() # ========================= # 12. 顯示迴歸結果 # ========================= print("\n==============================") print("OLS 迴歸分析結果") print("==============================") print(model.summary()) # ============================================================ # 13. 儲存清理後資料 # ============================================================ df.to_csv( "clean_aremos_data.csv", encoding="utf-8-sig" ) print("\n分析完成！") print( "清理後的資料已儲存為 clean_aremos_data.csv" )
